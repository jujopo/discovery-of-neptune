	\begin{frame}{The Mystery of Uranus}
		\begin{itemize}
			\item Newton’s laws accurately predicted most planetary motions.
			\item However, Uranus’s observed orbit deviated slightly from predictions.
			\item These deviations hinted at an unseen gravitational influence.
		\end{itemize}
		\vspace{1em}
		\alert{Could mathematics alone reveal the existence of a new planet?}
	\end{frame}
	
	% ---- Perturbation Theory ----
	\begin{frame}{Perturbation Theory}
		\begin{columns}
			\column{0.6\textwidth}
			\begin{block}{Concept}
				Perturbation theory allows us to approximate the behavior of a complex system by starting from a simpler, solvable one and adding corrections.
			\end{block}
			\begin{itemize}
				\item Direct perturbations: known causes → effects.
				\item Inverse perturbations: observed effects → unknown causes.
			\end{itemize}
			\column{0.4\textwidth}
			\includegraphics[width=\linewidth]{orbit_diagram.png}
		\end{columns}
	\end{frame}
	
	% ---- Mathematical Section ----
	\begin{frame}{Mathematical Formulation}
		\begin{align*}
			F &= ma \\
			&\Rightarrow m_U \ddot{\vec{r}}_U = -\frac{G m_S m_U}{r_U^3} \vec{r}_U + \sum_i \vec{F}_{\text{pert},i} \\
			\Delta r_U(t) &\approx f(\text{mass, distance, angle})
		\end{align*}
		\vspace{1em}
		\begin{block}{Goal}
			Given the residuals in Uranus’s orbit, determine the mass and position of the perturbing body.
		\end{block}
	\end{frame}
